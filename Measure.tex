    \begin{equation*}

% DEFINITIONS SECTION
\subsection{Definitions}

I will begin with the definitions of partitions, and upper and lower sums.

\begin{definition}[Partition]
    Let $[a, b] \subset \mathbb{R}$ then $P$, a partition, is defined as
    \begin{equation*}
        P = \{x_0, x_1, \ldots, x_n\}
    \end{equation*}
    Such that 
    \begin{equation*}
        a = x_0 < x_1 < \ldots < x_n = b
    \end{equation*}
\end{definition}

\begin{remark}
    P is a finite collection of points that subdivides the interval [a, b]. Importantly
    it is finite and partitions the domain of a function $f$ on $\mathbb{R}$.
\end{remark}

% UPPER SUMS
\begin{definition}[Upper sums]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$, the upper sums of
    $f$ for a partition $P$ are defined as
    \begin{equation*}
        U(f, P) = \sum_{k=1}^{n} (x_k - x_{k-1}) \sup_{[x_{k-1}, x_k]} f
    \end{equation*}
    Where $U(f)$ is defined as
    \begin{equation*}
        U(f) = \inf_P U(f, P)
    \end{equation*}
\end{definition}


% LOWER SUMS
\begin{definition}[Lower sums]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$, the lower sums of
    $f$ for a partition $P$ are defined as
    \begin{equation*}
        L(f, P) = \sum_{k=1}^{n} (x_k - x_{k-1}) \inf_{[x_{k-1}, x_k]} f
    \end{equation*}
    Where $L(f)$ is defined as 
    \begin{equation*}
        L(f) = \sup_P L(f, P)
    \end{equation*}
\end{definition}

Now defining Riemann Integration.

% RIEMANN INTEGRATION DEFINITION
\begin{definition}[Riemann Integration]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$.

    If for every $\epsilon > 0$ there exists a partition $P_\epsilon$ such that 
    \begin{equation}
        U(f, P_\epsilon) - L(f, P_\epsilon) < \epsilon
    \end{equation}

    then $f(x)$ is Riemann Integrable with 
    \begin{equation}
        \int f \, dx = U(f) = L(f)
    \end{equation}
\end{definition}


Notice from the definition that a function being Riemann Integrable relies on
us finding a partition in which the upper and lower sums converge. In general
how do we know before checking if a function will be Riemann Integrable?







% RIEMANN FAILURE SECTION
\subsection{Riemann Failure}


% FUNCTION
\subsubsection{Dirichlet Function}
Let the sequence of numbers $\{r_1, r_2, \ldots\}$ enumerate the rationals on $[0, 1]$.

Let $f_n(x)$ be a series of functions defined on $[0, 1]$ of the form
\begin{equation}
    f_n(x) = \begin{cases}
    1: & x \in \{r_1, r_2, \ldots, r_n\} \\
    0: & x \not\in \{r_1, r_2, \ldots, r_n\}
          \end{cases}
\end{equation}


% UPPER AND LOWER SUMS
\subsubsection{Riemann Integration}
I want to show each $f_n$ is Riemann integrable. From the definition I must show
\begin{equation*}
    U(f_n, P_n) - L(f_n, P_n) < \epsilon
\end{equation*}

for every $\epsilon > 0$, for some partition $P_n$.

To begin fix $n \in \mathbb{N}$.

Take the partition $P_n$ such that $(x_k - x_{k-1}) = \frac{\epsilon}{4n}$ for all consecutive
pairs of points in $P_n$. 
\begin{equation*}
    P = \{0, \frac{\epsilon}{4n}, \frac{2\epsilon}{4n}, \ldots, 1\}
\end{equation*}

The total number of intervals in which there may exist rational numbers is 2n;
since there are n rationals where each can be in at most 2 intervals at once.

\begin{remark}
    if $r_k = x_k$ for instance, then $[x_{k-1}, x_k]$ and $[x_k, x_{k+1}]$ both contains $r_k$
\end{remark}


From the definition of upper sums
\begin{equation*}
    U(f_n, P_n) = \sum_{k=1}^{n} (x_k - x_{k-1}) \sup_{[x_{k-1}, x_k]} f
\end{equation*}

The length of each interval as defined above is $(x_k - x_{k-1}) = (\frac{\epsilon}{4n})$, 
there exists at most $2n$ of these intervals, each with $\sup_{[x_k - x_{k-1}]} f = 1$. 

All other possible intervals have a max combined length of 1,
each with $\sup_{[x_k - x_{k-1}]} f = 0$.

This gives
\begin{equation*}
    U(f_n, P_n) = (\frac{\epsilon}{4n})(2n)(1) + (1)(0) = \frac{\epsilon}{2}
\end{equation*}

For the lower sums, since $\mathbb{I}$ is dense in $\mathbb{R}$, 
between any two real numbers there exists an irrational.

Meaning on any interval $[x_{k-1}, x_k]$, $\inf_{[x_{k-1}, x_k]} f = 0$.

This gives
\begin{equation*}
    L(f_n, P_n) = 0
\end{equation*}

This gives for any $\epsilon > 0$ for some $N \in \mathbb{N}$ whenever $n \geq N$
\begin{equation*}
    U(f_n, P_n) - L(f_n, P_n) = \frac{\epsilon}{2} < \epsilon
\end{equation*}

meaning the function is Riemann integrable with
\begin{equation*}
    \int_{0}^{1} f_n \, dx = L(f_n) = 0
\end{equation*}


% LIMIT
\subsubsection{Limit}
Letting $n \to \infty$ gives the limit of $f_n$ as 
\begin{equation}
    \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation}

Call this function $f$.

Now both $\mathbb{Q}$ dense in $\mathbb{R}$
as well as $\mathbb{I}$ dense in $\mathbb{R}$.

Meaning for any partition $P$, on any interval $[x_{k-1}, x_k] \subset [0, 1]$
there exists both a rational and irrational.

This gives $\inf_{[x_{k-1}, x_k]} f = 0$ and $\sup_{[x_k - x_{k-1}]} f = 1$ on every interval.

Which forces
\begin{equation*}
    U(f, P) = 1
\end{equation*}

and
\begin{equation*}
    L(f, P) = 0
\end{equation*}

for every partition $P$. Implying $U(f) = 1$ and $L(f) = 0$ which says that $f$ is
not Riemann integrable.


% LEBESGUE SAVIOR SECTION
\subsection{Lebesgue Explanation}

% MEASURE 0
\subsubsection{Measure 0}
To understand measure 0 first consider the concept of length.

\begin{definition}[Length]
    The length of an open interval $(a, b) \subset \mathbb{R}$ is defined as
    \begin{equation}
        l((a, b)) = b - a
    \end{equation}
\end{definition}

Now to define measure 0.

\begin{definition}[Measure 0]
    $A$ is a measure zero set if for every $\epsilon > 0$ there exists a countable covering $\bigcup_n O_n$ 
    where each $O_n$ is an open interval such that  

    \begin{equation}
        A \subset \bigcup_n O_n
    \end{equation}
    
    and
    \begin{equation}
        \sum_{n} l(O_n) \leq \epsilon
    \end{equation}
\end{definition}


% LEBESGUE CRITERION
\subsubsection{Lebesgue Criterion}

By the Lebesgue criterion
\begin{theorem}[Lebesgue Criterion]
    A bounded function $f\colon [a, b] \to \mathbb{R}$ is Riemann integrable iff the set of discontinuities 
    is measure 0
\end{theorem}

We have now defined the class of Riemann integrable functions in terms of a concept of measure
which did not previously exist in the context of Riemann integration.


% F DISCONTINUOUS EVERYWHERE
\subsubsection{Dirichlet Discontinuous Everywhere}

Recall that
\begin{equation*}
    f = \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation*}

I want to show that f is discontinuous on all of $[0, 1]$.

First observe the definition of continuity.

\begin{definition}
    A function f is continuous at a point c if for every $\epsilon > 0$
    there exists a $\delta > 0$ such that whenever $|x - c| < \delta$ it follows that
    $|f(x) - f(c)| < \epsilon$
\end{definition}

% DISCONTINUOUS
\paragraph{f Discontinuous}
Begin by fixing $\epsilon \leq 1$, I will take $\epsilon_0 = \frac{1}{2}$. Choose any point $c \in [0, 1]$.

Again both $\mathbb{Q}$ and $\mathbb{I}$ dense in $\mathbb{R}$.

First consider $c \in \mathbb{Q} \, \cap \, [0, 1]$ then for any $\delta$ chosen there exists some irrational point
\begin{equation}
    x \in (c - \delta, c + \delta) \implies |f(x) - f(c)| = 1 \geq \epsilon_0
\end{equation}

Consider the case $c \in \mathbb{I} \, \cap \, [0, 1]$ then for any $\delta$ chosen there exists some rational point
\begin{equation}
    x \in (c - \delta, c + \delta) \implies |f(x) - f(c)| = 1 \geq \epsilon_0
\end{equation}

Meaning the set of discontinuous points of $f$ is all of 
$(\mathbb{Q} \, \cap \, [0, 1]) \cup (\mathbb{I} \, \cap \, [0, 1]) = [0, 1]$ where $|[0, 1]| = 1$.
\begin{remark}
    I do not prove it here but it is proved in section 2.1
\end{remark}

% f_n DISCONTINUOUS
\paragraph{fn Measure 0 Discontinuities}
To show that the set of discontinuities for every $f_n$ is measure 0 first consider the set of
points at which $f_n$ is discontinuous, call it $D_n$.

For any points $c \in \{r_1, r_2, \ldots, r_n\}$ since $\mathbb{I}$ dense in $\mathbb{R}$
we have for a fixed $\epsilon < 1$ and any $\delta > 0$
\begin{equation*}
    |x - c| < \delta \implies |f(x) - f(c)| = 1 \geq \epsilon
\end{equation*}

Making 
\begin{equation*}
    D_n = \{r_1, r_2, \ldots, r_n\}
\end{equation*}

For any points $c \not\in \{r_1, r_2, \ldots, r_n\}$ we can take $\delta = \frac{\min(|x - r_1|, |x - r_2|,
\ldots, |x - r_n|)}{2}$ since there are a fixed amount of rationals for any fixed $n \in \mathbb{N}$.

This would give us for 
\begin{equation*}
    |x - c| < \delta \implies |f(x) - f(c)| = 0 < \epsilon
\end{equation*} 

making $f_n$ continuous on any points $x \not\in \{r_1, r_2, \ldots, r_n\}$. 

Now consider the open covering
$\bigcup_k O_k$ such that each 
\begin{equation*}
    O_k = (r_k - \frac{\epsilon}{2^{k+1}}, r_k + \frac{\epsilon}{2^{k+1}})
\end{equation*}

Then
\begin{align*}
    \sum_k l(O_k) &= \sum_k \frac{\epsilon}{2^{k}} \\
    &= \frac{\frac{\epsilon}{2}}{1 - \frac{1}{2}} \\
    &= \epsilon 
\end{align*}

giving 
\begin{equation*}
    \sum_k l(O_k) \leq \epsilon
\end{equation*} 

making $D_n$ measure 0 by definition.

We now understand why each $f_n$ integrates but $f$ itself fails to integrate.
Our understanding lays not in the language of the Riemann integral itself but instead in the
language of measure.  

We also see that the method by which we partition the domain fails for functions
that oscillate too quickly and will fail to be defined in spaces outside of $\mathbb{R}$.


\subsection{Lebesgue Savior}

Under mild assumptions, integrating with respect to measure can gives us
\begin{equation}
    \lim_{n \to \infty} \int f_n \, d\mu = 0 = \int f \, d\mu
\end{equation}





% OUTER MEASURE
\section{Outer Measure}
Let $|.|$ denote outer measure.

\begin{definition}[Outer Measure]
    $|A| = \inf \sum_k l(O_k)$ Where $\inf$ is taken over all open coverings such that
    $A \subset \bigcup_k O_k$
\end{definition}

\subsection{Properties of Outer Measure}


% PROPERTIES OF OUTER MEASURE
\begin{theorem}
\mbox{}
\begin{enumerate}
  \item If $A \subseteq B$ then $|A| \leq |B|$
  \item $|[a,b]| = b - a$
  \item If $A_1, A_2, \ldots$ satisfy $\bigcup_k A_k = B$, then $|B| \leq \sum_k |A_k|$
  \item $|A + t| = |A|$
\end{enumerate}
\end{theorem}

I will now prove each in order.

% MONOTONCITY
\begin{proof}[Proof: Monotoncity]
    For monotonicity (1) first observe that since 
    \begin{equation*}
        A \subset B \subset \bigcup_k O_k
    \end{equation*}

    any open covering of $B$ is an open covering of $A$.

    Let 
    \begin{equation*}
        S_a = \left\{ \sum_k l(O_k): O \text{ is an open cover of } A \right\}
    \end{equation*}
    Where each $O$ is an open cover of the form $\bigcup_k O_k$.

    Similarly let
    \begin{equation*}
        S_b = \left\{ \sum_k l(O_k): O \text{ is an open cover of } B \right\}
    \end{equation*}

    $S_b \subset S_a$ since any open covering of $B$ is an open covering of $A$.
    It then follows $\inf S_a \leq \inf S_b$.

    Giving $|A| \leq |B|$.

    \begin{remark}
        Say $B \subset C$ then $\inf C \leq \inf B$; since any lower bound of $C$ is
        a lower bound of $B$, making $\inf C$ a lower bound of $B$, where
        $\inf B$ is greatest lower bound.
    \end{remark}
\end{proof}


% AGREEMENT ON [a, b]
\begin{proof}[Proof: $\lvert [a, b{]}\rvert = b - a$]
    For the second property of outer measure begin by considering the open cover
    for $[a, b]$
    \begin{equation*}
        (a - \epsilon, a + \epsilon) \cup (a, b) \cup (b - \epsilon, b + \epsilon)
    \end{equation*}

    Since $|[a, b]|$ is $\inf$ over all coverings we have 
    \begin{equation*}
        |[a, b]| \leq 2\epsilon + b - a + 2\epsilon
    \end{equation*}
    Where $\epsilon$ is arbitrary giving 
    \begin{equation*}
        |[a, b]| \leq b - a
    \end{equation*}

    For the other direction consider any open covering of $[a, b]$. I want to show that
    \begin{equation*}
        \sum_k l(O_k) \geq b - a
    \end{equation*}

    hence giving 
    
    \begin{equation*}
        \inf \sum_k l(O_k) \geq b - a
    \end{equation*}
    
    Now $[a, b]$ is a compact interval, by Heine-Borel for every open cover
    \begin{equation*}
        [a, b] \subset \bigcup_k O_k
    \end{equation*}
    
    there exists a finite sub-collection such that
    \begin{equation*}
        [a, b] \subset \bigcup_{k=1}^{n} O_k
    \end{equation*}
    
    Where

    \begin{equation*}
        \sum_{k=1}^{n} l(O_k) \leq \sum_{k=1}^{\infty} l(O_k)
    \end{equation*}
    Since length is non negative. 

    Begin with $n=1$ intervals. Any interval containing $[a, b]$ ie
    $[a, b] \subset O_k$ has length $l(O_k) \geq b - a$,
    since 
    \begin{equation*}
        (a, b) \subset [a, b] \subset O_k
    \end{equation*}

    Suppose the claim holds for $n$ intervals.
    
    Now consider $n+1$ intervals covering $[a, b]$. 
    
    $b$ must be contained
    in at least one of these intervals, call that interval $(c, d)$. 
    If $c < a$ then $l(c, d) \geq b - a$.
    So consider $a < c < b < d$. Since $(c, d)$ covers none of $[a, c]$ we can invoke
    the inductive hypothesis on $[a, c]$ giving
    \begin{equation*}
        \sum_{k=1}^{n} l(O_k) \geq c - a
    \end{equation*} 
    for $O_k$ covering $[a, c]$. Adding this to $l(c, d)$ gives 
    \begin{align*}
        \sum_{k=1}^{n+1} l(O_k) &\geq (c - a) + (d - c) \\
        &= d - a \\
        &\geq b - a
    \end{align*}

    Completing the proof.
\end{proof}


% COUNTABLE SUBADDITIVTY
\begin{proof}[Proof: Countable Subadditivity]
    I want to show 
    \begin{equation*}
        |\bigcup_k A_k| \leq \sum_k |A_k|
    \end{equation*}

    Let
    \begin{equation*}
        A = \bigcup_k A_k
    \end{equation*}

    Since outer measure is $\inf$ over all open covers, take for each $A_k$ the open cover
    $O_k = \bigcup_i O_{ki}$ where
    \begin{equation*}
        \sum_i l(O_{ki}) < |A_k| + \frac{\epsilon}{2^{k+1}}
    \end{equation*}

    Now since the $O_k$ cover each $A_k$ and $A = \bigcup_k A_k$
    \begin{equation*}
        A \subset \bigcup_k O_k
    \end{equation*}

    Meaning
    \begin{align*}
        |A| &\leq \sum_k \sum_i l(O_{ki}) \\
        &< \sum_k |A_k| + \frac{\epsilon}{2^{k+1}} \\
        &= \sum_k |A_k| + \frac{\epsilon}{2} \\
    \end{align*}

    Which holds for all $\epsilon > 0$ giving

    \begin{equation*}
        |\bigcup_k A_k| \leq \sum_k |A_k|
    \end{equation*}

    Completing the proof.

\end{proof}


% TRANSLATION INVARIANCE
\begin{proof}[Proof: Translation Invariance]
    I would like to show $|A + t| = |A|$.

    First observe that 
    \begin{equation*}
        l((a, b)) = l((a + t, b + t))
    \end{equation*}
    
    since 
    \begin{equation*}
        l((a + t, b +t)) = (b + t) - (a + t) = b - a
    \end{equation*}

    Now observe that for any open cover of $A$ call it $O_A$, $O_{A+t}$ covers $A + t$. 
    
    Where
    $O_{A + t}$ is defined as 
    \begin{equation*}
        \bigcup_k (O_k + t)
    \end{equation*}
    
    for $O_k \in O_A$. 
    
    Take any point in
    $A + t$ call it y. Then $y = x + t$ for some $x \in A$. Where $x \in O_k$ for some $O_k \in O_A$.
    Then $y \in O_k + t$. Where 
    \begin{equation*}
        l(O_k) = l(O_k + t)
    \end{equation*}

    Making the set of covers of $A$ is a subset of the set of covers of $A + t$.

    Let 
    \begin{equation*}
        S_A = \left\{ \sum_k l(O_k): O \text{ is an open cover of } A \right\}
    \end{equation*}
    Where each $O$ is an open cover of the form $\bigcup_k O_k$.

    Similarly let
    \begin{equation*}
        S_{A + t} = \left\{ \sum_k l(O_k): O \text{ is an open cover of } A + t \right\}
    \end{equation*}

    Meaning since $|A| \inf$ over all covers ie $\inf S_A$ and $S_A \subset S_{A + t}$,
    it follows $|A + t| \leq |A|$.

    Now consider the sets $A + t + (-t)$ and $A + t$ where the left side is a translations of the right
    side. Giving 
    \begin{equation*}
        |A + t + (-t)| \leq |A + t| \implies |A| \leq |A + t|
    \end{equation*} 

    Which follows from the above. Completing the proof $|A + t| = |A|$.

\end{proof}


It seems that outer measure has all the desired properties we would want or need. The problem
comes when we want to integrate with respect to measure. We would like to break
up the set we are integrating over into disjoint sets whose union is the set itself. Therefore
we would additionally want to see.

\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}

Which is not much different then our property of sub-additivity. This may seem
intuitively true at first, but as we will see in a moment this is not actually true for
all sets. Therefore we would like to be explicit about where and when the
properties holds to be coherent in our application of measure theory.



% OUTER MEASURE FAILS
\subsection{Failure of Outer Measure as a Measure}

% EQUIVALENCE CLASSES
\subsubsection{Equivalence Classes}
\begin{definition}[Equivalence Relation]
    x is said to be equivalent to y denoted by $x \sim y$ iff $x - y \in \mathbb{Q}$
\end{definition}


To be an equivalence relation it must satisfy the following properties
\begin{enumerate}
    \item $x \sim x$
    \item If $x \sim y$ then $y \sim x$
    \item If $x \sim y$ and $y \sim z$ then $x \sim z$.
\end{enumerate}


\begin{proof}[Proof: Equivalence Relation]
    For $x \sim s$, notice for any value $x$, $x - x = 0$ and $0 \in \mathbb{Q}$.

    For $x \sim y \implies y \sim x$, notice that $y - x = -1(x - y)$ where $\mathbb{Q}$ is close under
    multiplication.

    For $x \sim y$ and $y \sim z$, then $x \sim z$, notice $(x - y) + (y - z) = (x - z)$
    where $\mathbb{Q}$ is closed under addition.
\end{proof}


\begin{definition}[Equivalence Class on $[0, 1{]}$]
    An equivalence class on $[0, 1]$ is a set of all equivalent elements $\{y: y \sim x\}$
    for all $x \in [0, 1]$
\end{definition}

\begin{lemma}
    Any two equivalence classes on $[0, 1]$ are disjoint.
\end{lemma}

\begin{proof}[Proof: Equivalence Classes Disjoint]
Suppose for contradiction the intersection of two equivalence classes was non empty.

Let there be some element in $A$ call it $z$ and the same element in $B$. 
For arbitrary $x \in A$ and $y \in B$, $x - z \in \mathbb{Q}$ and
$y - z \in \mathbb{Q}$ . Let $x - z = u$ and $y - z = v$.
\end{proof}

Solving for $x$ and $y$ gives $x = u - z$ and $y = v - z$. Now subtracting $x - y$, we get
\begin{equation*}
    x - y = (u - z) - (v - z) = u - v
\end{equation*}

But $u, v \in \mathbb{Q}$, where $\mathbb{Q}$ is closed
under subtraction, giving $x - y \in \mathbb{Q}$. Which says $x \sim y$. 

Taking any $x \in A$ gives $x \sim y$ for some $y \in B$ forcing $x \in B$. Making
$A \subset B$. Similarly taking any $y \in B$ gives $y \sim x$ for some $x \in A$. Making
$B \subset A$ or $A = B$. 

Therefore any two equivalence classes on $[0, 1]$ are disjoint.


% VITALLI
\subsubsection{Vitalli}

For the following argument I will use the Axiom of Choice to show that, countable additivity
does not necessarily hold for all subsets of $\mathbb{R}$.

% AXIOM OF CHOICE
\begin{axiom}
    For any collection of nonempty sets, there exists a function assigning to each
    set one of its elements.
\end{axiom}

To begin, let a set $V$ contain a single element from each equivalence class on $[0, 1]$. 
The equivalence classes on the interval $[0, 1]$ are of the form $(\mathbb{Q} + x) \cap [0, 1]$.
Making $V$ uncountable.

Now consider the translations $V + q$ for $q \in [-1, 1] \cap \mathbb{Q}$. 

\begin{lemma}
    For any $p, q \in [-1, 1] \cap \mathbb{Q}$, the sets $V + p$ and $V + q$ are disjoint.
\end{lemma}


% TRANSLATED CLASSES
\begin{proof}[Proof: Translated Classes Disjoint]
To see this suppose some element $w$ existed such that 
\begin{equation*}
    w \in (V + p) \cap (V + q)
\end{equation*}
for $p, q \in \mathbb{Q} \cap [0, 1]$. 

Then $w = v + p$ for some $v \in V$ and 
$w = u + q$ for some $u \in V$. So 
\begin{equation*}
    v - w = p - q
\end{equation*}
making $v - u \in \mathbb{Q}$.

Meaning $v \sim u$, but $V$ only contains one element from each equivalence class meaning $v = u$
forcing $p = q$. But that says the translated classes are the same, meaning each
translated class must be disjoint.
\end{proof}

Now I will show that the translated classes are bounded in measure but infinite summations of
values; a contradiction.

\begin{lemma}[Vitalli Bounded]
    $[0, 1] \subset \bigcup_q (V + q) \subset [-1, 2]$
\end{lemma}

\begin{proof}[Proof: Vitalli Bounded]
Take any arbitrary $x \in [0, 1]$.

$x$ sits in some equivalence class since $V$ contains some representative from that class.

Meaning $x - v = q$ for some $q \in \mathbb{Q}$ and some $v \in V$. 

Since both $x, v \in [0, 1]$ it implies that $q \in [-1, 1] \cap \mathbb{Q}$.
Since $\bigcup_q (V + q)$ is a union over all $q \in [-1, 1] \cap \mathbb{Q}$, we have

\begin{equation*}
    [0, 1] \subset \bigcup_q (V + q)
\end{equation*}

Now take an arbitrary $v \in V$ and $q \in [-1, 1] \cap \mathbb{Q}$.
$v \in [0, 1]$ and $q \in [-1, 1]$ forcing $v + q \in [-1, 2]$.

Meaning

\begin{equation*}
    \bigcup_q (V + q) \subset [-1, 2]
\end{equation*}

Completing the proof.
\end{proof}

Now by monotonicity it must be the case that
\begin{equation}
    |[0, 1]| \leq |\bigcup_q (V + q)| \leq |[-1, 2]|
\end{equation}

Which implies
\begin{equation}
    1 \leq |\bigcup_q (V + q)| \leq 3
\end{equation}

Since outer measure is translation invariant for any unique $p, q$
$|V + q| = |V + p|$.

Now suppose it is true for disjoint sets
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}

then
\begin{equation}
    |\bigcup_q (V + q)| = \sum_q |(V + q)|
\end{equation}

Suppose that $|(V + q)| > 0$, since there are an infinite amount of $q \in [-1, 1]$ it would force
$\sum_q |(V + q)| = \infty$. 

Suppose then that $|(V + q)| = 0$, it would then force $\sum_q |(V + q)| = 0$.

Measure is non negative meaning $|(V + q)| \geq 0$, but either gives a contradiction to
\begin{equation}
    1 \leq |\bigcup_q (V + q)| \leq 3
\end{equation}

Meaning for the vitalli set, (a valid subset of R from the Axiom of Choice)
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}
does not hold.

Meaning it is not true in general and therefore we can not assert that
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}

This as you will see later actually implies that outer measure is not a valid measure on
all subsets of R.


% INTEGRATION ON MEASURE
\section{Integration on Measure}

First I will define what it means to be a measure and where exactly
measures are defined.

Before that I will define a $\sigma$-algebra and measurable space.

% SIMGA ALGEBRA
\begin{definition}[$\sigma$-algebra]
    A $\sigma$-algebra on a set $X$ is a set of subsets of $X$ satisfying the
    following three properties.

    \begin{enumerate}
        \item $\null \in S$
        \item for any set $E \in S$, $X \ E \in S$
        \item for any sequence of sets $E_1, E_2, \ldots$ where each $E_k \in S$, $\bigcup_k E_k \in S$
    \end{enumerate}
\end{definition}

From the properties we can quickly derive $X \in S$. First observe
$X \ E$ is $\{x \in X: x \not\in E\}$. Then from the first two properties $X \in S$.

Additionally it follows from De Morgan's laws that for any $E_1, E_2, \ldots$ 
where each $E_k \in S$, $\bigcap_k E_k \in S$.

\begin{align*}
    E_k \in S &\implies E_k^{c} \in S \\
    &\implies \bigcup_k E_k^{c} \in S \\
    &\implies (\bigcup_k E_k^{c})^{c} \in S \\
    &\implies \bigcap_k E_k \in S
\end{align*}

Where the last line follows from
\begin{align*}
    (\bigcup_k E_k^{c}) &= (\bigcap_k E_k)^{c} \\
    &\implies (\bigcup_k E_k^{c})^{c} = ((\bigcap_k E_k)^{c})^{c} \\
    &\implies (\bigcup_k E_k^{c})^{c} = \bigcap_k E_k
\end{align*}

\begin{remark}
    Notice that $S$ is defined specifically such that it contains the sets for which
    we later want to apply countable additivity too.
\end{remark}


% MEASURE SPACE
I will now define a measure space.

\begin{definition}[Measure Space]
    A measure space is defined as a set $X$ and a $\sigma$-algebra 
    defined on the set $X$ denoted by $(X , S)$.
\end{definition}


% MEASURE DEFINITION
Now to define a measure and a measure space.

\begin{definition}[Measure Space]
    Let $(X, S)$ be a measurable space. $\mu$ is a measure on the measure space, denoted
    by $(X, S, \mu)$, if $\mu(\null) = 0$, and $\mu(\bigcup_k E_k) = \sum_k \mu(E_k)$ for disjoint
    $E_k$ where each $E_k$ in $S$. 
\end{definition}

Now we can define integration with respect to measure and prove our result from earlier.
First again I will define partitions and lower sums.

% S-PARITITONS
\begin{definition}[$S$-Parititions]
    Let $(X, S, \mu)$ be a measure space, then $P$ an $S$-Partitions is defined as
    \begin{equation}
        P = \{E_1, E_2, \ldots, E_n\}
    \end{equation}

    Where each of the $E_k$ are disjoint, $E_k \in S$, and $\bigcup_k^{n} = X$.
\end{definition}

\begin{remark}
    Notice again $P$ contains finite sets.
\end{remark}

%S-MEASRUABLE F
\begin{definition}[S-Measurable]
    Let $(X, S, \mu)$ be a measurable space. A function $f$ is $S$-measurable
    if $f^{-1}((a, \infty)) \in S$ for every real number $a$. 
\end{definition}


% LOWER SUMS
\begin{definition}[Lower sums]
    Let $(X, S, \mu)$ be a measure space.

    Let $f$ be an $S$-measurable function $f \colon X \to [0, \infty]$, the lower sums of
    $f$ for a $S$-Partition $P$ are defined as
    \begin{equation*}
        L(f, P) = \sum_{k=1}^{n} \mu(E_k) \inf_{E_k} f
    \end{equation*} 
\end{definition}

Now finally integration with respect to measure.

% LEBESEGUE MEASURE
\begin{definition}[Lebesgue Measure]
    Let $(X, S, \mu)$ be a measure space.
    \begin{equation}
        \int f \, d\mu = \sup_{P} L(f, P)
    \end{equation}
\end{definition}


Finally recall our function from earlier
\begin{equation*}
    f = \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation*}

Consider the $S$-Partition.
\begin{equation*}
    \{[0, 1]\, \cap \, \mathbb{Q}, [0, 1] \setminus \mathbb{Q}\}
\end{equation*}

From earlier 
\begin{equation*}
    [0, 1]\, \cap \, \mathbb{Q}
\end{equation*}

is measure 0, where

\begin{equation*}
    \inf_{[0, 1]\, \cap \, \mathbb{Q}} f = 1
\end{equation*}

forcing

\begin{equation*}
    [0, 1] \setminus \mathbb{Q}
\end{equation*}

to be measure 1 since we know $|[0, 1]| = 1$ and the two sets
union to $[0, 1]$.

Where
\begin{equation*}
    \inf_{[0, 1] \setminus \mathbb{Q}} f = 0
\end{equation*}

Integrating with respect to measure then we get the following equation
\begin{equation}
    \int f \, d\mu = (0)(1) + (1)(0)
\end{equation}

Where from left to right we have, measure 0 times inf 1 plus
measure 1 times inf 0.

\section{Conclusion}
I began with Riemann integration failing to exist at the point wise limit
of Riemann integrable functions. To explain why this was the case I used the language
of measure. I then showed outer measure which had almost all the properties we wanted
failed to be a measure on $\mathbb{R}$. I then defined $\sigma$-algebras and measure
space in which countable additivity is valid. Finally I showed that our original function
could be handled by integrating with respect to measure, where we did not
need to use intervals to integrate but only rather the concept of measure. 

In general measure gives us the ability to go outside of $\mathbb{R}$ while keeping
all of the properties we are looking for at the cost of limiting exactly where our
methods hold.
