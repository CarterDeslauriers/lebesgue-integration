    \end{equation*}
    Where each $E_k \in S$ and 
    \begin{equation*}
        \bigcup_k E_k = X
    \end{equation*}
    
    and
    \begin{equation*}
        \mathbb{L}(f, P) = \sum_k \mu(E_k)\inf_{E_k} f
    \end{equation*}


\end{definition}

% DEFINITIONS SECTION
\subsection{Definitions}

I will begin with the definitions of partitions, and upper and lower sums.

\begin{definition}[Partition]
    Let $[a, b] \subset \mathbb{R}$ then $P$, a partition, is defined as
    \begin{equation*}
        P = \{x_0, x_1, \ldots, x_n\}
    \end{equation*}
    Such that 
    \begin{equation*}
        a = x_0 < x_1 < \ldots < x_n = b
    \end{equation*}
\end{definition}

\begin{remark}
    P is a finite collection of points that subdivides the interval [a, b]. Importantly
    it is finite and partitions the domain of a function $f$ on $\mathbb{R}$.
\end{remark}

% UPPER SUMS
\begin{definition}[Upper sums]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$, the upper sums of
    $f$ for a partition $P$ are defined as
    \begin{equation*}
        U(f, P) = \sum_{k=1}^{n} (x_k - x_{k-1}) \sup_{[x_{k-1}, x_k]} f
    \end{equation*}
    Where $U(f)$ is defined as
    \begin{equation*}
        U(f) = \inf_P U(f, P)
    \end{equation*}
\end{definition}


% LOWER SUMS
\begin{definition}[Lower sums]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$, the lower sums of
    $f$ for a partition $P$ are defined as
    \begin{equation*}
        L(f, P) = \sum_{k=1}^{n} (x_k - x_{k-1}) \inf_{[x_{k-1}, x_k]} f
    \end{equation*}
    Where $L(f)$ is defined as 
    \begin{equation*}
        L(f) = \sup_P L(f, P)
    \end{equation*}
\end{definition}

Now defining Riemann Integration.

% RIEMANN INTEGRATION DEFINITION
\begin{definition}[Riemann Integration]
    Let $f$ be a bounded function $f \colon [a, b] \to \mathbb{R}$.

    If for every $\epsilon > 0$ there exists a partition $P_\epsilon$ such that 
    \begin{equation}
        U(f, P_\epsilon) - L(f, P_\epsilon) < \epsilon
    \end{equation}

    then $f(x)$ is Riemann Integrable with 
    \begin{equation}
        \int f = U(f) = L(f)
    \end{equation}
\end{definition}


Notice from the definition that a function being Riemann Integrable relies on
us finding a partition in which the upper and lower sums converge. In general
how do we know before checking if a function will be Riemann Integrable?







% RIEMANN FAILURE SECTION
\subsection{Riemann Failure}


% FUNCTION
\subsubsection{Dirichlet Function}
Let the sequence of numbers $\{r_1, r_2, \ldots\}$ enumerate the rationals on $[0, 1]$.

Let $f_n(x)$ be a series of functions defined on $[0, 1]$ of the form
\begin{equation}
    f_n(x) = \begin{cases}
    1: & x \in \{r_1, r_2, \ldots, r_n\} \\
    0: & x \not\in \{r_1, r_2, \ldots, r_n\}
          \end{cases}
\end{equation}


% UPPER AND LOWER SUMS
\subsubsection{Riemann's Sums}
I want to show each $f_n$ is Riemann integrable. From the definition I must show
\begin{equation*}
    U(f_n, P_n) - L(f_n, P_n) < \epsilon
\end{equation*}

for every $\epsilon > 0$, for some partition $P_n$.

To begin fix $n \in \mathbb{N}$.

Take the partition $P_n$ such that $(x_k - x_{k-1}) = \frac{\epsilon}{4n}$ for all consecutive
pairs of points in $P_n$. 
\begin{equation*}
    P = \{0, \frac{\epsilon}{4n}, \frac{2\epsilon}{4n}, \ldots, 1\}
\end{equation*}

The total number of intervals in which there may exist rational numbers is 2n;
since there are n rationals where each can be in at most 2 intervals at once.

\begin{remark}
    if $r_k = x_k$ for instance then $[x_{k-1}, x_k]$ and $[x_k, x_{k+1}]$ both contains $r_k$
\end{remark}


From the definition of upper sums
\begin{equation*}
    U(f_n, P_n) = \sum_{k=1}^{n} (x_k - x_{k-1}) \sup_{[x_{k-1}, x_k]} f
\end{equation*}

The length of each interval as defined above is $(x_k - x_{k-1}) = (\frac{\epsilon}{4n})$, 
there exists at most $2n$ of these intervals, each with $\sup_{[x_k - x_{k-1}]} f = 1$. 

All other possible intervals have a max combined length of 1,
each with $\sup_{[x_k - x_{k-1}]} f = 0$.

This gives
\begin{equation*}
    U(f_n, P_n) = (\frac{\epsilon}{4n})(2n)(1) + (1)(0) = \frac{\epsilon}{2}
\end{equation*}

For the lower sums since $\mathbb{I}$ is dense in $\mathbb{R}$, 
between any two real numbers there exists an irrational.

Meaning on any interval $[x_{k-1}, x_k]$, $\inf_{[x_{k-1}, x_k]} f = 0$.

This gives
\begin{equation*}
    L(f_n, P_n) = 0
\end{equation*}

This gives for any $\epsilon > 0$ for some $N \in \mathbb{N}$ whenever $n \geq N$
\begin{equation*}
    U(f_n, P_n) - L(f_n, P_n) = \frac{\epsilon}{2} < \epsilon
\end{equation*}

meaning the function is Riemann integrable with
\begin{equation*}
    \int_{0}^{1} f_n = L(f_n) = 0
\end{equation*}


% LIMIT
\subsubsection{Limit}
Letting $n \to \infty$ gives the limit of $f_n$ as 
\begin{equation}
    \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation}

Call this function $f$.

Now both $\mathbb{Q}$ dense in $\mathbb{R}$
as well as $\mathbb{I}$ dense in $\mathbb{R}$.

Meaning for any partition $P$, on any interval $[x_{k-1}, x_k] \in P$
there exists both a rational and irrational.

This gives $\inf_{[x_{k-1}, x_k]} f = 0$ and $\sup_{[x_k - x_{k-1}]} f = 1$ on every interval.

Which forces
\begin{equation*}
    U(f, P) = 1
\end{equation*}

and
\begin{equation*}
    L(f, P) = 0
\end{equation*}

for every partition $P$. Implying $U(f) = 1$ and $L(f) = 0$ which says that $f$ is
not Riemann integrable.


% LEBESGUE SAVIOR SECTION
\subsection{Lebesgue Explanation}

% MEASURE 0
\subsubsection{Measure 0}
To understand measure 0 first consider the concept of length.

\begin{definition}[Length]
    The length of an open interval $(a, b) \subset \mathbb{R}$ is defined as
    \begin{equation}
        l((a, b)) = b - a
    \end{equation}
\end{definition}

Now to define measure 0.

\begin{definition}[Measure 0]
    $A$ is a measure zero set if for every $\epsilon > 0$ there exists a countable covering $\bigcup_n O_n$ 
    where each $O_n$ is an open interval such that  

    \begin{equation}
        A \subset \bigcup_n O_n
    \end{equation}
    
    and
    \begin{equation}
        \sum_{n} l(O_n) \leq \epsilon
    \end{equation}
\end{definition}


% LEBESGUE CRITERION
\subsubsection{Lebesgue Criterion}

By the Lebesgue criterion
\begin{theorem}[Lebesgue Criterion]
    A bounded function $f\colon [a, b] \to \mathbb{R}$ is Riemann integrable iff the set of discontinuities 
    is measure 0
\end{theorem}

We have now defined the class of Riemann integrable functions in terms of a concept of measure
which did not previously exist in the context of Riemann integration.

Recall that

\begin{definition}

\paragraph{fn Measure 0 Discontinuities}
To show that the set of discontinuities for every $f_n$ is measure 0 first consider the set of

For any points $c \in \{r_1, r_2, \ldots, r_n\}$ since $\mathbb{I}$ dense in $\mathbb{R}$
we have for a fixed $\epsilon < 1$ and any $\delta > 0$
\begin{equation*}
    |x - c| < \delta \implies |f(x) - f(c)| = 1 \geq \epsilon
\end{equation*}

Making 
\begin{equation*}
    D_n = \{r_1, r_2, \ldots, r_n\}
\end{equation*}

For any points $c \not\in \{r_1, r_2, \ldots, r_n\}$ we can take $\delta = \frac{\min(|x - r_1|, |x - r_2|,

\end{equation*} 

making $f_n$ continuous on any points $x \not\in \{r_1, r_2, \ldots, r_n\}$. 

Now consider the open covering
$\bigcup_k O_k$ such that each 
\begin{equation*}

Then
\begin{align*}
    \sum_k l(O_k) &= \sum_k \frac{\epsilon}{2^{k}} \\
    &= \frac{\frac{\epsilon}{2}}{1 - \frac{1}{2}} \\
    &= \epsilon 
\end{align*}

giving 
\begin{equation*}
    \sum_k l(O_k) \leq \epsilon
\end{equation*} 

making $D_n$ measure 0 by definition.

We know understand why each $f_n$ integrates but $f$ itself fails to integrate.
Our understanding lays not in the language of the Riemann integral itself but instead in the

% OUTER MEASURE
\section{Outer Measure}

\begin{definition}[Outer Measure]
    $|(A)| = \inf \sum_k l(O_k)$ Where $\inf$ is taken over all open coverings such that
    $A \subset \bigcup_k O_k$
\end{definition}

\subsection{Agrees eveyrwhere}
Outer measure agrees with length on open sets and [a, b] = (a, b) in terms of measure.
Translation invaraince, subadditivity, monotoncity


% PROPERTIES OF OUTER MEASURE
\begin{theorem}
\begin{enumerate}
  \item If $A \subseteq B$ then $|A| \leq |B|$
  \item $|[a,b]| = b - a$
  \item If $A_1, A_2, \ldots$ satisfy $\bigcup_k A_k = B$, then $|B| \leq \sum_k |A_k|$
  \item $|A + t| = |A|$
\end{enumerate}
\end{theorem}

I will now prove each in order.

% MONOTONCITY
\begin{proof}[Proof: Monotoncity]
    For monotonicity (1) first observe that since since 
    \begin{equation*}
    A \subset B \subset \bigcup_k O_k
    \end{equation*}

    any open covering of $B$ is an open covering of $A$.

    Now outer measure is defined as $\inf$ over all open coverings. Therefore there exists an
    open covering $\bigcup_k O_k$, such that 
    \begin{equation*}
        \bigcup_k l(O_k) < |B| + \epsilon
    \end{equation*}
    Since any open covering
    of $B$ is an open covering of $A$ 
    \begin{equation*}
        |A| \leq \sum_k l(O_k) < |B| + \epsilon
    \end{equation*} 
    Where $\epsilon$ is arbitrary $> 0$. Meaning $|A| \leq |B|$
\end{proof}

% AGREEMENT ON [a, b]
\begin{proof}[Proof: $\lvert [a, b{]}\rvert = b - a$]
    For the second property of outer measure consider the open cover 
    for $[a, b]$ 
    \begin{equation*}
        (a - \epsilon, a + \epsilon) \cup (a, b) \cup (b - \epsilon, b + \epsilon)
    \end{equation*}

    Since $|[a, b]|$ is $\inf$ over all coverings we have 
    \begin{equation*}
        |[a, b]| \leq 2\epsilon + b - a + 2\epsilon
    \end{equation*}
    Where $\epsilon$ is arbitrary giving 
    \begin{equation*}
        |[a, b]| \leq b - a
    \end{equation*}

    For the other direction consider any open covering of $[a, b]$. I want to show that
    \begin{equation*}
        \sum_k l(O_k) \geq b - a
    \end{equation*}

    hence giving 
    
    \begin{equation*}
        \inf \sum_k l(O_k) \geq b - a
    \end{equation*}
    
    Now $[a, b]$ is a compact interval, by Heine-Borel for every open cover
    \begin{equation*}
        [a, b] \subset \bigcup_k O_k
    \end{equation*}
    
    there exists a finite subcollection such that
    \begin{equation*}
        [a, b] \subset \bigcup_{k=1}^{n} O_k
    \end{equation*}
    
    Where

    \begin{equation*}
        \sum_{k=1}^{n} l(O_k) \leq \sum_{k=1}^{\infty} l(O_k)
    \end{equation*}
    Since length is non negative. 
    
    Suppose the claim holds for $n$ intervals.

    Begin with $n=1$ intervals. Any interval containing $[a, b]$ ie
    $[a, b] \subset O_k$ has length $l(O_k) \geq b - a$,
    since 
    \begin{equation*}
        (a, b) \subset [a, b] \subset O_k
    \end{equation*}
    
    Now consider $n+1$ intervals covering $[a, b]$. 
    
    $b$ must be contained
    in at least one of these intervals call it $(c, d)$. If $c < a$ then $l(c, d) >= b - a$
    so consider $a < c < b < d$; since $(c, d)$ covers none of $[a, c]$ we can invoke
    the inductive hypothesis 
    \begin{equation*}
        \sum_{k=1}^{n} l(O_k) \geq c - a
    \end{equation*} 
    for $O_k$ covering $[a, c]$. Adding this to $l(c, d)$ gives 
    \begin{align*}
        \sum_{k=1}^{n+1} l(O_k) &\geq (c - a) + (d - c) \\
        &= d - a \\
        &\geq b - a
    \end{align*}

    Completing the proof.
\end{proof}


% COUNTABLE SUBADDITIVTY
\begin{proof}[Proof: Countable Subadditivity]
    I want to show 
    \begin{equation*}
        |\bigcup_k A_k| \leq \sum_k |A_k|
    \end{equation*}

    Let
    \begin{equation*}
        A = \bigcup_k A_k
    \end{equation*}

    Since outer measure is $\inf$ over all open covers, take for each $A_k$ the open cover
    \begin{equation*}
        O_k < |A_k| + \frac{\epsilon}{2^{k+1}}
    \end{equation*}

    Now since the $O_k$ cover $A$
    \begin{equation*}
        A \subset \bigcup_k O_k
    \end{equation*}

    giving
    \begin{align*}
        |A| &\leq |\bigcup_k O_k| \\
        &= \sum_k |O_k| \\
        &= \sum_k \frac{\epsilon}{2^{k+1}} \\
        &= \frac{\epsilon}{2}
    \end{align*}

    Since this is true for any arbitary $\epsilon$ and $A$ is $\inf$ over all coverings
    we get 
    \begin{equation*}
        |\bigcup_k A_k| \leq \sum_k |A_k|
    \end{equation*}

    Completing the proof.

\end{proof}


% TRANSLATION INVARIANCE
\begin{proof}[Proof: Translation Invariance]
    I would like to show $|A + t| = |A|$.

    First observe that 
    \begin{equation*}
        l((a, b)) = l((a + t, b + t))
    \end{equation*}
    
    since 
    \begin{equation*}
        l((a + t, b +t)) = (b + t) - (a + t) = b - a
    \end{equation*}

    Now observe that for any open cover of $A$ call it $O_A$, $O_{A+t}$ covers $A$. 
    
    Where
    $O_{A + t}$ is defined as 
    \begin{equation*}
        \bigcup_k (O_k + t)
    \end{equation*}
    
    for $O_k \in O_A$. 
    
    Take any point in
    $A + t$ call it y. Then $y = x + t$ for some $x \in A$. Where $x \in O_k$ for some $O_k \in O_A$.
    Then $y \in O_k + t$. Where 
    \begin{equation*}
        l(O_k) = l(O_k + t)
    \end{equation*}

    Making the set of covers of $A$, call them $S \subset S_t$, where $S_t$ is the set of open
    covers of $A + t$. 
    
    Meaning since $|A| \inf$ over all covers ie $\inf S$ and $S \subset S_t$,
    it follows $|A + t| \leq |A|$.
    \begin{remark}
        Say $B \subset C$ then $\inf C \leq \inf B$; since taking $\inf$ over larger set is smaller. 
    \end{remark}

    Now consider the sets $A + t + (-t)$ and $A + t$ where the left side is a translations of the right
    side. Giving 
    \begin{equation*}
        |A + t + (-t)| \leq |A + t| \implies |A| \leq |A + t|
    \end{equation*} 

    Which follows from the above. Completing the proof $|A + t| = |A|$.

\end{proof}

% EQUIVALENCE CLASSES
\subsubsection{Equivalence Classes}
\begin{definition}[Equavalince Relation]
    x is said to be equivalent to y denoted by $x \sim y$ iff $x - y \in \mathbb{Q}$
\end{definition}


To be an eqvaaulence relation it must satisfy $x \sim x$, if $x \sim y$
then $y \sim x$, and if $x \sim y$ and $y \sim z$ then $x \sim z$. 


\begin{proof}[Proof: Equavalice Relation]
    For the first, notice for any value $x$, $x - x = 0$ and $0 \in \mathbb{Q}$.

    For the second, notice that $y - x = -1(x - y)$ where $\mathbb{Q}$ is close under
    multiplication.

    For the third, notice $(x - y) + (y - z) = (x - z)$ where $\mathbb{Q}$ is closed under
    addition.
\end{proof}


\begin{definition}[Equavalence Class on [0, 1]]
    An equvalence class on $[0, 1]$ is a set of all equivalent elements ${y: y \sim x}$ 
    for all $x \in [0, 1]$
\end{definition}


Now any two equivalence classes on $[0, 1]$ are disjoint. To see this
suppose the intersection of two equivalence classes was non empty. So let there be some element
in $A$ call it $z$ and the same element in $B$. By definition $x - z \in \mathbb{Q}$ and
$y - z \in \mathbb{Q}$ for arbitrary $x \in A$ and $y \in B$. Let $x - z = u$ and $y - z = v$. 
Solving for $x$ and $y$ gives $x = u - z$ and $y = v - z$. Now if we subtract $x - y$, we get
$x - y = (u + z) - (v + z) = u - v$. But $u, v \in \mathbb{Q}$ where $\mathbb{Q}$ is closed
under subtraction giving $x - y \in \mathbb{Q}$, but that says $x \sim y$. 

Then taking any $x \in A$ gives $x \sim y$ for some $y \in B$ forcing $x \in B$. Making
$A \subset B$. Similarly taking any $y \in B$ gives $y \sim x$ for some $x \in A$. Making
$B \subset A$ or $A = B$. Therefore any two equivalence classes on $[0, 1]$ are disjoint.


% VITALLI
\subsubsection{Vitalli}

Let V contain a single element from each equivalence class. There exists uncountably many
of these classes in which we have no way of choosing, but where we can
still choose and element from each from the Axiom of Choice.

Now for every $q \in [-1, 1] \cap \mathbb{Q}$ the sets $V + q$ are disjoint. To see
this suppose some element $w$ existed such that $w \in (V + p) \cap (V + q)$ for
$p, q \in \mathbb{Q} \cap [0, 1]$. Then $w = v + p$ for some $v \in V$ and 
$w = u + q$ for some $u \in V$. So $v - w = p - q$ making $v - u \in \mathbb{Q}$ meaning
$v \sim u$ but $V$ only contains one element from each equivalence class meaning $v = u$
forcing $p = q$. But that says the translated classes are the same, meaning each
translated class must be disjoint.

Now 
\begin{equation}
    [0, 1] \subset \bigcup_q (V + q) \subset [-1, 2]
\end{equation}

Take any arbitrary $x \in [0, 1]$,
$x$ sits in some equivalence class where $V$ contains some representative from that class.
Giving $x - v \in \mathbb{Q}$. Since $x, v \in [0, 1] \implies x - v \in [-1, 1] \cap \mathbb{Q}$.
Where every $q \in [-1, 1]$ is used. Now take an arbitrary $v + q$ simply
$v \in [0, 1]$ and $q \in [-1, 1]$ forcing $v + q \in [-1, 2]$. 

Now by monotonicity it must be the case that
\begin{equation}
    |[0, 1]| \leq |\bigcup_q (V + q)| \leq [-1, 2]
\end{equation}

Which implies
\begin{equation}
    1 \leq |\bigcup_q (V + q)| \leq 3
\end{equation}

Since each $V + q$ consists of exactly the same amount of elements
$|V + q| = |V + p|$.

Now suppose it is true for disjoint sets
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}

then
\begin{equation}
    |\bigcup_q (V + q)| = \sum_q |(V + q)|
\end{equation}

Suppose that $|(V + q)| > 0$, since there are an infinite amount of $q \in [-1, 1]$ it would force
$\sum_q |(V + q)| = \infty$. 

Suppose then that $|(V + q)| = 0$, it would then force $\sum_q |(V + q)| = 0$.

But measure is non negative meaning $|(V + q)| \geq 0$ but ether gives a contradiction to
\begin{equation}
    1 \leq |\bigcup_q (V + q)| \leq 3
\end{equation}

Meaning for the vitalli set, (a valid subset of R implying the axiom of choice)
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}
does not hold.

Meaning it is not true in general and there we can not assert that
\begin{equation}
    |\bigcup_k A_k| = \sum_k |A_k|
\end{equation}

This as you will see later actually implies that outer measure is not a valid measure on
all subsets of R.
