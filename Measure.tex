\begin{definition}[Pointwise Convergence]
For every $\epsilon > 0$ there exists some $N \in \mathbb{N}$ such that whenever $n \geq N$ it

    |f_n(x) - f(x)| < \epsilon

\begin{align*}

\begin{align*}

\end{align*}

Meaning our function converges pointwise to $f(x)$ for $f(x) = 0$

Consider both methods of integration and there definitions

% RIEMANN INTEGRATION DEFINTION
\begin{definition}[Reimann Integration]
    Let $f$ be a bounded function $f: [a, b] \to \mathbb{R}$.
    Let $P$ be a partition of the domain of the form
    \begin{equation*}
        P = \{x_0, x_1, \ldots, x_n\}
    \end{equation*}
    Such that 
    \begin{equation*}
        a = x_0 < x_1 < \ldots < x_n = b
    \end{equation*}
    If for every $\epsilon > 0$ there exists a partion $P_e$ 
    such that $U(f, P_\epsilon) - L(f, P_\epsilon) < \epsilon$ then $f(x)$ is
    Riemann Integrable with $\int f = U(f) = L(f).$

    Where 
    \begin{equation*}
        U(f, P_\epsilon) = \sum_{k=1}^{n} (x_k - x_{k-1}) \sup_{[x_{k-1}, x_k]} f
    \end{equation*}
    \begin{equation*}
        L(f, P_\epsilon) = \sum_{k=1}^{n} (x_k - x_{k-1}) \inf_{[x_{k-1}, x_k]} f
    \end{equation*}
    
    and
    \begin{equation*}
        U(f) = \inf_P U(f, P)
    \end{equation*}
    \begin{equation*}
        L(f) = \sup_P L(f, P)
    \end{equation*}

\end{definition}


% LEBESEGUE INTEGRATION DEFINITION
\begin{definition}[Lebesgue Integration]
    Let $(X, S, \mu)$ be a measure space and let $f(x) \to \mathbb{R}$
    then $\int f = \sup L(f, P)$.

    Where P is an $S$-Partition. Where an $S$-Partition is of the form
    \begin{equation*}
        P = {E_1, E_2, \ldots, E_n}
    \end{equation*}
    Where each $E_k \in S$ and 
    \begin{equation*}
        \bigcup_k E_k = X
    \end{equation*}
    
    and
    \begin{equation*}
        \mathbb{L}(f, P) = \sum_k \mu(E_k)\inf_{E_k} f
    \end{equation*}


\end{definition}

% RIEMANN FAILURE SECTION
\subsection{Riemann Failure}


% FUNCTION
\subsubsection{Function}
Let the sequence of rational numbers $\{r_1, r_2, \ldots\}$ enumerate the rationals on $[0, 1]$.

Let $f_n(x)$ be a series of functions of the form
\begin{equation}
    f_n(x) = \begin{cases}
    1: & x \in \{r_1, r_2, \ldots, r_n\} \\
    0: & x \not\in \{r_1, r_2, \ldots, r_n\}
          \end{cases}
\end{equation}


% UPPER AND LOWER SUMS
\subsubsection{Upper and Lower Sums}
Fix $n \in \mathbb{N}$.

Take the partition $P_n$ such that $(x_k - x_{k-1}) = \frac{\epsilon}{4n}$ for all consecutive
pairs of points in $P_n$. 

The total number of intervals in which there are rationals numbers is 2n;
since there are n rationals where each can be in at most 2 intervals at once.

\begin{remark}
    if $r_k = x_k$ for instance then $[x_{k-1}, x_k]$ and $[x_k, x_{k+1}]$ both contains $r_k$
\end{remark}


This gives
\begin{equation*}
    U(f_n, P_n) = (\frac{\epsilon}{4n})(2n)(1) + (1)(0) = \frac{\epsilon}{2}
\end{equation*}

Where each length $(x_k - x_{k-1}) = (\frac{\epsilon}{4n})$, there exists at most $2n$ of these intervals, 
each with $\sup_{[x_k - x_{k-1}]} f = 1$. All other possible intervals have a max combined length of 1,
each with $\sup_{[x_k - x_{k-1}]} f = 0$.

Since $\mathbb{I}$ is dense in $\mathbb{R}$, between any two real numbers there exists an irrational.
Meaning on any interval $[x_{k-1}, x_k]$, $\inf_{[x_{k-1}, x_k]} f = 0$ giving
\begin{equation*}
    L(f_n, P_n) = 0
\end{equation*}

This gives for any $\epsilon > 0$ for some $N \in \mathbb{N}$ whenever $n \geq N$ it follows
\begin{equation*}
    U(f_n, P_n) - L(f_n, P_n) = \frac{\epsilon}{2} < \epsilon
\end{equation*}

meaning the function is Riemann integrable with
\begin{equation*}
    \int_{0}^{1} f_n = L(f_n) = 0
\end{equation*}


% LIMIT
\subsubsection{Limit}
Since the sequence of values $r_k$ are countable we have 
\begin{equation}
    \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation}

Call this function $f$

Now for any partition $P$, since both $\mathbb{Q}$ is dense in $\mathbb{R}$
as well as $\mathbb{I}$ dense in $\mathbb{R}$, on any interval $[x_{k-1}, x_k] \in P$
there exists both a rational and irrational.

This gives $\inf_{[x_{k-1}, x_k]} f = 0$ as well $\sup_{[x_k - x_{k-1}]} f = 1$ on every interval.
Which forces
\begin{equation*}
    U(f, P) = 1
\end{equation*}
and
\begin{equation*}
    L(f, P) = 0
\end{equation*}

for every partition $P$ implying $U(f) = 1$ and $L(f) = 0$ which says that $f$ is
not Riemann integrable.


% LEBESGUE SAVIOR SECTION
\subsection{Lebesgue Explanation}

% MEASURE 0
\subsubsection{Measure 0}

\begin{definition}[Measure 0]
    $A$ is a measure zero set if for every $\epsilon > 0$ there exists a countable covering $\bigcup_n O_n$ 
    where each $O_n$ is an open interval such that  

    \begin{equation}
        A \subset \bigcup_n O_n
    \end{equation}
    
    Where 
    \begin{equation}
        \sum_{n} l(O_n) \leq \epsilon
    \end{equation}

    and l() denotes the length of a open interval defined
    as $b - a$ for any open interval $(a, b)$.
\end{definition}


% LEBESGUE CRITERION
\subsubsection{Lebesgue Criterion}

By the Lebesgue criterion
\begin{theorem}[Lebesgue Criterion]
    A bounded function $f\colon [a, b] \to \mathbb{R}$ is Riemann integrable iff the set of discounties 
    is Measure 0
\end{theorem}


% F DISCOUNTIOUS EVERYWHERE
\subsubsection{f discontinous everywhere}

Recall that
\begin{equation*}
    \lim_{n \to \infty} f_n = \begin{cases}
                        1: & x \in \mathbb{Q} \cap [0, 1] \\
                        0: & x \not\in \mathbb{Q} \cap [0, 1]
                        \end{cases}
\end{equation*}

I want to show that f is discontinous on all of $[0, 1]$
First observe the definition of continuity.

\begin{definition}
    A function f is continuous at a point c if for every $\epsilon > 0$
    there exists a $\delta > 0$ such that whenever $|x - c| < \delta$ it follows that
    $|f(x) - f(c)| < \epsilon$
\end{definition}

Now fix $\epsilon_0 \leq 1$ take $\epsilon_0 = \frac{1}{2}$. Choose any point $c \in [0, 1]$.
Again both $\mathbb{Q}$ and $\mathbb{I}$ dense in $\mathbb{R}$.

First consider $c \in \mathbb{Q}$ then for any $\delta$ chosen there exists some irrational point
\begin{equation}
    x \in (c - \delta, c + \delta) \implies |f(x) - f(c)| = 1 \geq \epsilon_0
\end{equation}

Consider the case $c \in \mathbb{I}$ then for any $\delta$ chosen there exists some rational point
\begin{equation}
    x \in (c - \delta, c + \delta) \implies |f(x) - f(c)| = 1 \geq \epsilon_0
\end{equation}

Meaning the set of discontinuous points of $f$ is all of $[0, 1]$

To show that the set of discontinuities for every $f_n$ is measure 0 first consider the points of
discontinuity. From the section above $f_n$ is discontinuous at every $x \in \{r_1, r_2, \ldots, r_n\}$.
For any points $x \not\in \{r_1, r_2, \ldots, r_n\}$ we can take $\delta = \frac{\min(|x - r_1|, |x - r_2|,
\ldots, |x - r_n|)}{2}$ since there are a fixed amount of rationals for any fixed $n \in \mathbb{N}$.
This would give us for $|x - c| < \delta \implies |f(x) - f(c)| = 0 < \epsilon$ making $f_n$ continuous
on any points $x \not\in \{r_1, r_2, \ldots, r_n\}$. 

Now consider the open covering
$\bigcup_k O_k$ such that each $O_k = (r_k - \frac{\epsilon}{2^{k+1}}, r_k + \frac{\epsilon}{2^{k+1}})$.
Call $D_n$ the set of discontinoutis of $f_n$. Then $D_n \subset \bigcup_k O_k$ where
\begin{align*}
    \sum_k l(O_k) &= \sum_k \frac{\epsilon}{2^{k}} \\
    &= \frac{\frac{\epsilon}{2}}{1 - \frac{1}{2}} \\
    &= \epsilon 
\end{align*}

giving $\sum_k l(O_k) \leq \epsilon$ making $D_n$ measure 0 by definition.
